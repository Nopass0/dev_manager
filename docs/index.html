03 Lua Scripting

15 modules: HTTP, files, processes, services, UI automation, JSON, TCP.

-- HTTP + assertions
local resp = http.get("http://localhost:8080/health")
assert(resp.status == 200)

-- Project context
local services = dm_ctx.services()
local api = dm_ctx.service("api")

-- Process interaction
local p = proc_io.spawn("./app")
p:write("input\n")
local out = p:read_line()
p:wait()
